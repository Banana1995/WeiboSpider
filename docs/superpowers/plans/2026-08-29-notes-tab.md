```python
    def get_annotated_tweets(self):
        """Return non-deleted tweets that have at least one annotation,
        all platforms, ordered by created_at DESC.

        Used by the 笔记 tab to show tweets that have 划线评论.
        """
        with self._lock:
            rows = self.conn.execute(
                "SELECT * FROM tweets WHERE deleted = 0 "
                "AND id IN (SELECT DISTINCT tweet_id FROM annotations) "
                "ORDER BY created_at DESC"
            ).fetchall()
            results = []
            for row in rows:
                d = dict(row)
                d['pic_urls'] = json.loads(d.get('pic_urls', '[]') or '[]')
                d['retweet_pic_urls'] = json.loads(d.get('retweet_pic_urls', '[]') or '[]')
                d['is_retweet'] = bool(d.get('is_retweet'))
                d['deleted'] = bool(d.get('deleted'))
                results.append(d)
            return results
```

```js
function switchTab(tab) {
  currentTab = tab;
  document.querySelectorAll('.tab').forEach(t => t.classList.toggle('active', t.dataset.tab === tab));
  const isLogs = tab === 'logs';
  const isPs = tab === 'ps';
  const isNotes = tab === 'notes';
  const isXq = tab === 'xueqiu';
  const hideBatch = isLogs || isPs || isNotes || isXq;
  const hidePagination = isLogs || isPs || isNotes;
  $('select-all-row').style.display = hideBatch ? 'none' : '';
  $('btn-batch-delete').style.display = hideBatch ? 'none' : (tab === 'trash' ? 'none' : '');
  $('btn-batch-restore').style.display = hideBatch ? 'none' : (tab === 'trash' ? '' : 'none');
  $('cards-container').style.display = isLogs ? 'none' : '';
  $('loader').style.display = isLogs ? 'none' : '';
  $('pagination').style.display = hidePagination ? 'none' : '';
  $('logs-container').style.display = isLogs ? 'block' : 'none';
  $('cards-container').classList.toggle('ps-mode', isPs || isNotes);
  if (isLogs) {
    loadLogs(1);
  } else if (isPs) {
    selected.clear();
    updateBatchBtn();
    loadPs();
  } else if (isNotes) {
    selected.clear();
    updateBatchBtn();
    loadNotes();
  } else {
    selected.clear();
    updateBatchBtn();
    reloadList();
  }
}
```

```js
async function loadNotes() {
  $('loader').classList.remove('hidden');
  try {
    const resp = await fetch('/api/notes');
    const tweets = await resp.json();
    $('cards-container').innerHTML = '';
    if (!tweets || tweets.length === 0) {
      $('cards-container').innerHTML = '<div style="text-align:center;padding:40px;color:#aaa;font-size:13px;">暂无笔记（还没有添加过划线评论）</div>';
    } else {
      tweets.forEach(t => renderCard(t));
    }
  } catch(e) { console.error(e); }
  $('loader').classList.add('hidden');
}
```
